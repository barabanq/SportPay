<?php

declare(strict_types=1);

namespace App\Services;

use App\Enums\TransactionStatus;
use App\Enums\TransactionType;
use App\Models\User;
use App\Models\WalletTransaction;
use DomainException;
use Illuminate\Support\Facades\DB;

class WalletService
{
    public function deposit(int $userId, float $amount, ?int $orderId = null): WalletTransaction
    {
        if ($amount <= 0) {
            throw new DomainException('Сумма пополнения должна быть больше нуля.');
        }

        return DB::transaction(function () use ($userId, $amount, $orderId): WalletTransaction {
            $user = User::where('id', $userId)->lockForUpdate()->firstOrFail();

            return $this->applyBalanceChange($user, $amount, TransactionType::Deposit, $orderId);
        });
    }

    public function withdraw(int $userId, float $amount): WalletTransaction
    {
        if ($amount <= 0) {
            throw new DomainException('Сумма вывода должна быть больше нуля.');
        }

        return DB::transaction(function () use ($userId, $amount): WalletTransaction {
            $user = User::where('id', $userId)->lockForUpdate()->firstOrFail();

            if ((float) $user->wallet_balance < $amount) {
                throw new DomainException('Недостаточно средств на балансе для вывода.');
            }

            return $this->applyBalanceChange($user, -$amount, TransactionType::Withdraw);
        });
    }

    /**
     * Единая точка изменения баланса и записи аудиторского следа (DRY).
     * Вызывается внутри активной БД-транзакции с заблокированным User.
     */
    public function applyBalanceChange(
        User $lockedUser,
        float $signedAmount,
        TransactionType $type,
        ?int $orderId = null
    ): WalletTransaction {
        $newBalance = round((float) $lockedUser->wallet_balance + $signedAmount, 2);

        if ($newBalance < 0) {
            throw new DomainException('Недостаточно средств на кошельке.');
        }

        $lockedUser->update(['wallet_balance' => $newBalance]);

        return WalletTransaction::create([
            'user_id' => $lockedUser->id,
            'order_id' => $orderId,
            'amount' => $signedAmount,
            'type' => $type,
            'balance_after' => $newBalance,
            'status' => TransactionStatus::Success,
        ]);
    }
}

<?php

declare(strict_types=1);

namespace App\Services;

use App\Enums\OrderStatus;
use App\Enums\TransactionType;
use App\Models\Lot;
use App\Models\Message;
use App\Models\Order;
use App\Models\Review;
use App\Models\User;
use DomainException;
use Illuminate\Support\Facades\DB;

class EscrowService
{
    private const PLATFORM_FEE_RATE = 0.05;

    public function __construct(
        private readonly WalletService $walletService
    ) {}

    public function purchaseLot(int $buyerId, int $lotId, int $quantity = 1): Order
    {
        return DB::transaction(function () use ($buyerId, $lotId, $quantity): Order {
            $lot = Lot::where('id', $lotId)->lockForUpdate()->firstOrFail();
            $buyer = User::where('id', $buyerId)->lockForUpdate()->firstOrFail();

            if ($lot->seller_id === $buyer->id) {
                throw new DomainException('Нельзя купить собственный лот.');
            }

            if ($lot->stock_quantity < $quantity) {
                throw new DomainException('Товара нет в нужном количестве.');
            }

            $totalPrice = round((float) $lot->price * $quantity, 2);

            $lot->decrement('stock_quantity', $quantity);

            $initialStatus = $lot->hasAutoDelivery()
                ? OrderStatus::Shipped
                : OrderStatus::Paid;

            $order = Order::create([
                'lot_id' => $lot->id,
                'buyer_id' => $buyer->id,
                'seller_id' => $lot->seller_id,
                'quantity' => $quantity,
                'unit_price' => $lot->price,
                'total_price' => $totalPrice,
                'status' => $initialStatus,
            ]);

            $this->walletService->applyBalanceChange(
                $buyer,
                -$totalPrice,
                TransactionType::Purchase,
                $order->id
            );

            $this->logSystemMessage(
                $order,
                $buyer->id,
                "Покупатель {$buyer->username} оплатил заказ. Средства ({$totalPrice} ₽) заморожены гарантом SportPay."
            );

            if ($lot->hasAutoDelivery()) {
                $this->logSystemMessage(
                    $order,
                    $lot->seller_id,
                    "⚡ АВТОВЫДАЧА ТОВАРА: {$lot->auto_delivery_text}"
                );
            }

            return $order;
        });
    }

    public function confirmOrder(int $orderId, int $buyerId, ?int $rating = null, ?string $comment = null): Order
    {
        return DB::transaction(function () use ($orderId, $buyerId, $rating, $comment): Order {
            $order = Order::where('id', $orderId)->lockForUpdate()->firstOrFail();

            if ($order->buyer_id !== $buyerId) {
                throw new DomainException('Только покупатель может подтвердить выполнение заказа.');
            }

            if (!$order->status->isConfirmable()) {
                throw new DomainException('Заказ уже закрыт или отменен.');
            }

            $seller = User::where('id', $order->seller_id)->lockForUpdate()->firstOrFail();
            $sellerPayout = round((float) $order->total_price * (1 - self::PLATFORM_FEE_RATE), 2);

            $order->update(['status' => OrderStatus::Completed]);

            $this->walletService->applyBalanceChange(
                $seller,
                $sellerPayout,
                TransactionType::Payout,
                $order->id
            );

            if ($rating !== null) {
                $this->recordReview($order, $buyerId, $seller, $rating, $comment);
            }

            $this->logSystemMessage(
                $order,
                $buyerId,
                "Покупатель подтвердил выполнение заказа. Продавцу выплачено {$sellerPayout} ₽ (комиссия биржи 5%)."
            );

            return $order;
        });
    }

    public function openDispute(int $orderId, int $actorId, string $reason): Order
    {
        return DB::transaction(function () use ($orderId, $actorId, $reason): Order {
            $order = Order::where('id', $orderId)->lockForUpdate()->firstOrFail();
            $actor = User::findOrFail($actorId);

            if (!$order->isParticipant($actor)) {
                throw new DomainException('Только участник сделки может открыть спор.');
            }

            if (!$order->status->isDisputable()) {
                throw new DomainException('По данному заказу спор уже открыт или сделка завершена.');
            }

            $order->update(['status' => OrderStatus::Disputed]);

            $this->logSystemMessage(
                $order,
                $actor->id,
                "⚠️ {$actor->username} открыл арбитражный спор! Причина: «{$reason}». Выплата заморожена, к сделке подключен арбитр SportPay."
            );

            return $order;
        });
    }

    public function resolveDispute(int $orderId, int $arbiterId, string $decision, ?string $comment = null): Order
    {
        return DB::transaction(function () use ($orderId, $arbiterId, $decision, $comment): Order {
            $arbiter = User::findOrFail($arbiterId);

            if (!$arbiter->isArbiter()) {
                throw new DomainException('Только официальный арбитр SportPay может выносить решение по спору.');
            }

            $order = Order::where('id', $orderId)->lockForUpdate()->firstOrFail();

            if ($order->status !== OrderStatus::Disputed) {
                throw new DomainException('Решение можно вынести только по заказу в статусе спора.');
            }

            $verdictNote = $comment ? " Комментарий арбитра: «{$comment}»." : '';

            if ($decision === 'refund_buyer') {
                $buyer = User::where('id', $order->buyer_id)->lockForUpdate()->firstOrFail();

                Lot::where('id', $order->lot_id)->increment('stock_quantity', $order->quantity);
                $order->update(['status' => OrderStatus::Cancelled]);

                $this->walletService->applyBalanceChange(
                    $buyer,
                    (float) $order->total_price,
                    TransactionType::Deposit,
                    $order->id
                );

                $this->logSystemMessage(
                    $order,
                    $arbiter->id,
                    "⚖️ Арбитр {$arbiter->username} решил спор в пользу покупателя. Средства ({$order->total_price} ₽) возвращены покупателю.{$verdictNote}"
                );

                return $order;
            }

            if ($decision === 'release_seller') {
                $seller = User::where('id', $order->seller_id)->lockForUpdate()->firstOrFail();
                $sellerPayout = round((float) $order->total_price * (1 - self::PLATFORM_FEE_RATE), 2);

                $order->update(['status' => OrderStatus::Completed]);

                $this->walletService->applyBalanceChange(
                    $seller,
                    $sellerPayout,
                    TransactionType::Payout,
                    $order->id
                );

                $this->logSystemMessage(
                    $order,
                    $arbiter->id,
                    "⚖️ Арбитр {$arbiter->username} решил спор в пользу продавца. Выплата ({$sellerPayout} ₽) переведена продавцу.{$verdictNote}"
                );

                return $order;
            }

            throw new DomainException('Неизвестный тип решения арбитража.');
        });
    }

    public function cancelAndRefundOrder(int $orderId, int $actorId): Order
    {
        return DB::transaction(function () use ($orderId, $actorId): Order {
            $order = Order::where('id', $orderId)->lockForUpdate()->firstOrFail();

            if ($order->buyer_id === $actorId && $order->status !== OrderStatus::Disputed) {
                throw new DomainException('Покупатель не может отменить оплаченный заказ напрямую. Откройте спор или попросите продавца сделать возврат.');
            }

            if (!$order->status->isConfirmable()) {
                throw new DomainException('Невозможно вернуть средства по завершенному заказу.');
            }

            $buyer = User::where('id', $order->buyer_id)->lockForUpdate()->firstOrFail();

            Lot::where('id', $order->lot_id)->increment('stock_quantity', $order->quantity);
            $order->update(['status' => OrderStatus::Cancelled]);

            $this->walletService->applyBalanceChange(
                $buyer,
                (float) $order->total_price,
                TransactionType::Deposit,
                $order->id
            );

            $this->logSystemMessage(
                $order,
                $actorId,
                "Заказ отменен продавцом. Средства ({$order->total_price} ₽) полностью возвращены покупателю."
            );

            return $order;
        });
    }

    private function recordReview(Order $order, int $buyerId, User $seller, int $rating, ?string $comment): void
    {
        Review::create([
            'order_id' => $order->id,
            'author_id' => $buyerId,
            'seller_id' => $seller->id,
            'lot_id' => $order->lot_id,
            'rating' => $rating,
            'comment' => $comment ?: 'Сделка прошла отлично!',
        ]);

        $avgRating = Review::where('seller_id', $seller->id)->avg('rating') ?? 5.00;
        $seller->update(['rating' => round((float) $avgRating, 2)]);
    }

    private function logSystemMessage(Order $order, int $userId, string $text): void
    {
        Message::create([
            'order_id' => $order->id,
            'user_id' => $userId,
            'body' => $text,
            'is_system' => true,
        ]);
    }
}

class ChatController extends Controller
{
    public function index(Request $request): View
    {
        $user =$this->currentUser();

        $orders = Order::with(['lot', 'buyer', 'seller', 'latestMessage'])
            ->where(function ($query) use ($user): void {
                $query->where('buyer_id',$user->id)
                    ->orWhere('seller_id', $user->id);
            })
            ->withCount([
                'messages as unread_count' => function ($query) use ($user): void {
                    $query->where('user_id', '!=',$user->id)->whereNull('read_at');
                },
            ])
            ->latest('updated_at')
            ->get();

        $activeOrder = null;

        if ($request->filled('order_id')) {$activeOrder = Order::with(['lot', 'buyer', 'seller', 'messages.user'])
                ->findOrFail((int) $request->integer('order_id'));

            $this->authorizeAction('view',$activeOrder);
        } elseif ($orders->isNotEmpty()) {$activeOrder = Order::with(['lot', 'buyer', 'seller', 'messages.user'])
                ->find($orders->first()->id);
        }

        if ($activeOrder !== null) {
            Message::where('order_id', $activeOrder->id)
                ->where('user_id', '!=', $user->id)
                ->whereNull('read_at')
                ->update(['read_at' => now()]);
        }

        return view('chat.index', [
            'user' => $user,
            'allUsers' => User::orderBy('id')->get(),
            'orders' => $orders,
            'activeOrder' => $activeOrder,
        ]);
    }

    public function messages(Request $request, Order$order): JsonResponse
    {
        $user =$this->currentUser();
        $this->authorizeAction('view',$order);

        $afterId = (int)$request->query('after_id', 0);

        $messages = Message::with('user')
            ->where('order_id', $order->id)
            ->where('id', '>', $afterId)
            ->oldest('id')
            ->get();

        if ($messages->isNotEmpty()) {
            Message::where('order_id', $order->id)
                ->where('user_id', '!=', $user->id)
                ->whereNull('read_at')
                ->update(['read_at' => now()]);
        }

        return response()->json([
            'messages' => $messages->map(fn (Message$msg): array => [
                'id' => $msg->id,
                'order_id' => $msg->order_id,
                'user_id' => $msg->user_id,
                'username' => $msg->user->username ?? 'Система',
                'body' => $msg->body,
                'is_system' => (bool) $msg->is_system,
                'time' => $msg->created_at?->format('H:i') ?? now()->format('H:i'),
            ]),
        ]);
    }

    public function store(Request $request, Order$order): JsonResponse|RedirectResponse
    {
        $user =$this->currentUser();
        $this->authorizeAction('sendMessage',$order);

        $data =$request->validate([
            'body' => ['required', 'string', 'max:2000'],
        ]);

        $message = Message::create([
            'order_id' => $order->id,
            'user_id' => $user->id,
            'body' => $data['body'],
            'is_system' => false,
        ]);

        $message->load('user');$order->touch();

        try {
            broadcast(new OrderMessageSent($message))->toOthers();
        } catch (\Throwable) {
            // Игнорируем ошибку сокета, если Reverb не запущен — отработает фоновый пуллинг
        }

        $payload = [
            'id' => $message->id,
            'order_id' => $order->id,
            'user_id' => $user->id,
            'username' => $user->username,
            'body' => $message->body,
            'is_system' => false,
            'time' => $message->created_at?->format('H:i') ?? now()->format('H:i'),
        ];

        if ($request->wantsJson()) {
            return response()->json([
                'status' => 'success',
                'message' => $payload,
            ]);
        }

        return back();
    }
}

use App\Services\WalletService;
use DomainException;
use Illuminate\Contracts\View\View;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;

class WalletController extends Controller
{
    public function __construct(
        private readonly WalletService $walletService
    ) {}

    public function index(): View
    {
        $user = $this->currentUser();

        $transactions = WalletTransaction::where('user_id', $user->id)
            ->latest('id')
            ->limit(50)
            ->get();

        $activeStatuses = [
            OrderStatus::Paid,
            OrderStatus::Shipped,
            OrderStatus::Disputed,
        ];

        // Сумма, которую пользователь заморозил как покупатель
        $frozenAsBuyer = (float) Order::where('buyer_id', $user->id)
            ->whereIn('status', $activeStatuses)
            ->sum('total_price');

        // Ожидаемая выплата (минус 5% комиссии) по активным продажам
        $pendingAsSeller = round((float) Order::where('seller_id', $user->id)
            ->whereIn('status', $activeStatuses)
            ->sum('total_price') * 0.95, 2);

        return view('wallet.index', [
            'user' => $user,
            'allUsers' => User::orderBy('id')->get(),
            'transactions' => $transactions,
            'frozenAsBuyer' => $frozenAsBuyer,
            'pendingAsSeller' => $pendingAsSeller,
        ]);
    }

    public function deposit(Request $request): RedirectResponse
    {
        $data = $request->validate([
            'amount' => ['required', 'numeric', 'min:100', 'max:500000'],
        ]);

        try {
            $this->walletService->deposit($this->currentUser()->id, (float) $data['amount']);

            return back()->with('success', 'Баланс успешно пополнен!');
        } catch (DomainException $e) {
            return back()->with('error', $e->getMessage());
        }
    }

    public function withdraw(Request $request): RedirectResponse
    {
        $data = $request->validate([
            'amount' => ['required', 'numeric', 'min:100', 'max:500000'],
        ]);

        try {
            $this->walletService->withdraw($this->currentUser()->id, (float) $data['amount']);

            return back()->with('success', 'Заявка на вывод средств выполнена!');
        } catch (DomainException $e) {
            return back()->with('error', $e->getMessage());
        }
    }
}

<?php

namespace App\Http\Requests\Auth;

use Illuminate\Auth\Events\Lockout;
use Illuminate\Contracts\Validation\ValidationRule;
use Illuminate\Foundation\Http\FormRequest;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;

class LoginRequest extends FormRequest
{
    /**
     * Determine if the user is authorized to make this request.
     */
    public function authorize(): bool
    {
        return true;
    }

    /**
     * Get the validation rules that apply to the request.
     *
     * @return array<string, ValidationRule|array<mixed>|string>
     */
    public function rules(): array
    {
        return [
            'email' => ['required', 'string', 'email'],
            'password' => ['required', 'string'],
        ];
    }

    /**
     * Attempt to authenticate the request's credentials.
     *
     * @throws ValidationException
     */
    public function authenticate(): void
    {
        $this->ensureIsNotRateLimited();

        if (! Auth::attempt($this->only('email', 'password'), $this->boolean('remember'))) {
            RateLimiter::hit($this->throttleKey());

            throw ValidationException::withMessages([
                'email' => trans('auth.failed'),
            ]);
        }

        RateLimiter::clear($this->throttleKey());
    }

    /**
     * Ensure the login request is not rate limited.
     *
     * @throws ValidationException
     */
    public function ensureIsNotRateLimited(): void
    {
        if (! RateLimiter::tooManyAttempts($this->throttleKey(), 5)) {
            return;
        }

        event(new Lockout($this));

        $seconds = RateLimiter::availableIn($this->throttleKey());

        throw ValidationException::withMessages([
            'email' => trans('auth.throttle', [
                'seconds' => $seconds,
                'minutes' => ceil($seconds / 60),
            ]),
        ]);
    }

    /**
     * Get the rate limiting throttle key for the request.
     */
    public function throttleKey(): string
    {
        return Str::transliterate(Str::lower($this->string('email')).'|'.$this->ip());
    }
}

namespace App\Policies;

use App\Enums\OrderStatus;
use App\Models\Order;
use App\Models\User;

class OrderPolicy
{
    /**
     * Просматривать сделку и читать чат могут покупатель, продавец и арбитр.
     */
    public function view(User $user, Order $order): Response
    {
        return $order->isParticipant($user)
            ? Response::allow()
            : Response::deny('У вас нет доступа к этой сделке.');
    }

    /**
     * Писать в чат сделки могут только участники (покупатель, продавец и арбитр).
     */
    public function sendMessage(User $user, Order $order): Response
    {
        return $order->isParticipant($user)
            ? Response::allow()
            : Response::deny('У вас нет доступа к чату этой сделки.');
    }

    /**
     * Подтвердить выполнение заказа может только покупатель.
     */
    public function confirm(User $user, Order $order): Response
    {
        return $order->buyer_id === $user->id
            ? Response::allow()
            : Response::deny('Только покупатель может подтвердить выполнение заказа.');
    }

    /**
     * Добровольно отменить заказ без спора может только продавец (или арбитр при споре).
     */
    public function cancel(User $user, Order $order): Response
    {
        if ($order->seller_id === $user->id || $user->isArbiter()) {
            return Response::allow();
        }

        if ($order->buyer_id === $user->id && $order->status === OrderStatus::Disputed) {
            return Response::allow();
        }

        return Response::deny('Покупатель не может отменить оплаченный заказ напрямую. Откройте спор.');
    }

    /**
     * Открыть спор может покупатель или продавец.
     */
    public function dispute(User $user, Order $order): Response
    {
        return ($order->buyer_id === $user->id || $order->seller_id === $user->id)
            ? Response::allow()
            : Response::deny('Только участник сделки может открыть спор.');
    }

    /**
     * Вынести решение по спору может только официальный арбитр.
     */
    public function resolve(User $user, Order $order): Response
    {
        return $user->isArbiter()
            ? Response::allow()
            : Response::deny('Только официальный арбитр SportPay может выносить решение по спору.');

        'product_type',
        'description',
        'auto_delivery_text',
        'price',
        'stock_quantity',

    ];

        'auto_delivery_text',
    ];

    public function hasAutoDelivery(): bool
    {

    }

    protected function casts(): array
    {
        return [
            'price' => 'decimal:2',
            'stock_quantity' => 'integer',
            'status' => LotStatus::class,
        ];
    }

    {
        return $this->belongsTo(User::class, 'seller_id');
    }

    public function orders(): HasMany
    {
        return $this->hasMany(Order::class, 'lot_id');
    }

    public function reviews(): HasMany
    {
        return $this->hasMany(Review::class, 'lot_id');
    }

    /**
     * Фильтрация активных лотов по категории и поисковой строке
     */
    public function scopeFilter(Builder $query, ?string $type, ?string $search): Builder
    {
        return $query
            ->where('lots.stock_quantity', '>', 0)
            ->where('lots.status', LotStatus::Active)
            ->when($type, fn (Builder $q) => $q->where('lots.product_type', $type))
            ->when($search, function (Builder $q) use ($search): void {
                $term = '%' . trim($search) . '%';
                $q->where(function (Builder $sub) use ($term): void {
                    $sub->where('lots.title', 'like', $term)
                        ->orWhere('lots.description', 'like', $term);
                });
            });
    }

    /**
     * Безопасная сортировка по цене, рейтингу продавца или новизне
     */
    public function scopeSortBy(Builder $query, ?string $sort): Builder
    {
        return match ($sort) {
            'price_asc' => $query->orderBy('lots.price', 'asc'),
            'price_desc' => $query->orderBy('lots.price', 'desc'),
            'rating_desc' => $query
                ->join('users', 'lots.seller_id', '=', 'users.id')
                ->select('lots.*')
                ->orderByDesc('users.rating')
                ->orderByDesc('lots.id'),
            default => $query->latest('lots.id'),
        };
    }

<script>
(function () {
    const box = document.getElementById('order-messages-box');
    const form = document.getElementById('order-chat-form');
    const input = document.getElementById('order-chat-input');
    const currentUserId = {{ (int) $user->id }};
    const orderId = {{ (int) $order->id }};
    const pollUrl = "{{ route('orders.messages.index', $order) }}";

    let lastMessageId = {{ (int) ($order->messages->last()?->id ?? 0) }};

    function forceScrollBottom() {
        if (!box) return;
        box.scrollTop = box.scrollHeight + 9999;
    }

    forceScrollBottom();
    window.addEventListener('load', () => {
        forceScrollBottom();
        setTimeout(forceScrollBottom, 120);
    });

    function renderOrderMessage(msg) {
        const msgId = Number(msg.id || 0);
        if (msgId > 0 && box.querySelector(`[data-msg-id="${msgId}"]`)) {
            return;
        }
        if (msgId > lastMessageId) {
            lastMessageId = msgId;
        }

        const emptyPlaceholder = document.getElementById('empty-order-chat');
        if (emptyPlaceholder) emptyPlaceholder.remove();

        const wrapper = document.createElement('div');
        if (msgId > 0) wrapper.setAttribute('data-msg-id', String(msgId));

        if (msg.is_system) {
            wrapper.className = 'p-3 rounded-xl bg-amber-500/10 border border-amber-500/20 text-center text-xs text-amber-300';
            wrapper.textContent = '🛡️ ' + msg.body;
        } else {
            const isMe = Number(msg.user_id) === currentUserId;
            wrapper.className = `flex flex-col ${isMe ? 'items-end' : 'items-start'}`;

            const meta = document.createElement('div');
            meta.className = 'text-[11px] text-slate-400 mb-1';
            meta.textContent = `${msg.username} • ${msg.time}`;

            const bubble = document.createElement('div');
            bubble.className = `max-w-md px-4 py-2.5 rounded-2xl text-sm ${isMe ? 'bg-emerald-500 text-slate-950 font-medium' : 'bg-slate-800 text-slate-100 border border-slate-700'}`;
            bubble.textContent = msg.body;

            wrapper.appendChild(meta);
            wrapper.appendChild(bubble);
        }

        box.appendChild(wrapper);
        forceScrollBottom();
    }

    form.addEventListener('submit', async function (e) {
        e.preventDefault();
        const body = input.value.trim();
        if (!body) return;

        input.value = '';
        input.focus();

        try {
            const response = await fetch(form.action, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'X-CSRF-TOKEN': '{{ csrf_token() }}',
                    'Accept': 'application/json'
                },
                body: JSON.stringify({ body })
            });

            if (response.ok) {
                const data = await response.json();
                if (data.message) renderOrderMessage(data.message);
            }
        } catch (err) {
            console.error('Ошибка отправки:', err);
        }
    });

    if (window.Echo) {
        window.Echo.channel(`orders.${orderId}`)
            .listen('.order.message.sent', (data) => renderOrderMessage(data));
    }

    setInterval(async () => {
        try {
            const res = await fetch(`${pollUrl}?after_id=${lastMessageId}`, {
                headers: { 'Accept': 'application/json' }
            });
            if (res.ok) {
                const data = await res.json();
                if (Array.isArray(data.messages)) {
                    data.messages.forEach(renderOrderMessage);
                }
            }
        } catch (e) {}
    }, 2000);
})();
