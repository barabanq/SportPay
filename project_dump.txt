<?php

declare(strict_types=1);

namespace App\Services;

use App\Enums\TransactionType;
use App\Models\User;
use App\Models\WalletTransaction;
use DomainException;
use Illuminate\Support\Facades\DB;

class WalletService
{
    public function deposit(User|int $userId, float $amount): WalletTransaction
    {
        if ($amount <= 0) {
            throw new DomainException('Сумма пополнения должна быть больше нуля.');
        }

        $targetUserId = $userId instanceof User ? $userId->id : $userId;

        return DB::transaction(function () use ($targetUserId, $userId, $amount): WalletTransaction {
            /** @var User $lockedUser */
            $lockedUser = User::whereKey($targetUserId)->lockForUpdate()->firstOrFail();

            if (! $lockedUser->status->canTrade()) {
                throw new DomainException('Ваш аккаунт заморожен или заблокирован.');
            }

            $newBalance = round((float) $lockedUser->wallet_balance + $amount, 2);
            $lockedUser->update(['wallet_balance' => $newBalance]);

            /** @var WalletTransaction $tx */
            $tx = WalletTransaction::create([
                'user_id' => $lockedUser->id,
                'order_id' => null,
                'type' => TransactionType::Deposit,
                'amount' => round($amount, 2),
                'balance_after' => $newBalance,
            ]);

            if ($userId instanceof User) {
                $userId->refresh();
            }

            return $tx;
        });
    }

    public function withdraw(User|int $userId, float $amount): WalletTransaction
    {
        if ($amount <= 0) {
            throw new DomainException('Сумма вывода должна быть больше нуля.');
        }

        $targetUserId = $userId instanceof User ? $userId->id : $userId;

        return DB::transaction(function () use ($targetUserId, $userId, $amount): WalletTransaction {
            /** @var User $lockedUser */
            $lockedUser = User::whereKey($targetUserId)->lockForUpdate()->firstOrFail();

            if (! $lockedUser->status->canTrade()) {
                throw new DomainException('Вывод средств недоступен для вашего статуса аккаунта.');
            }

            if ((float) $lockedUser->wallet_balance < $amount) {
                throw new DomainException('Недостаточно средств для вывода.');
            }

            $newBalance = round((float) $lockedUser->wallet_balance - $amount, 2);
            $lockedUser->update(['wallet_balance' => $newBalance]);

            /** @var WalletTransaction $tx */
            $tx = WalletTransaction::create([
                'user_id' => $lockedUser->id,
                'order_id' => null,
                'type' => TransactionType::Withdraw,
                'amount' => -round($amount, 2),
                'balance_after' => $newBalance,
            ]);

            if ($userId instanceof User) {
                $userId->refresh();
            }

            return $tx;
        });
    }
}

<?php

declare(strict_types=1);

namespace App\Services;

use App\Enums\LotStatus;
use App\Enums\OrderStatus;
use App\Enums\TransactionType;
use App\Models\Lot;
use App\Models\Message;
use App\Models\Order;
use App\Models\Review;
use App\Models\User;
use App\Models\WalletTransaction;
use DomainException;
use Illuminate\Support\Facades\DB;

class EscrowService
{
    public const PLATFORM_FEE_RATE = 0.05;

    public function purchaseLot(User|int $buyerId, Lot|int $lotId, int $quantity = 1): Order
    {
        if ($quantity < 1) {
            throw new DomainException('Количество должно быть не менее 1 шт.');
        }

        $resolvedBuyerId = $buyerId instanceof User ? $buyerId->id : $buyerId;
        $resolvedLotId = $lotId instanceof Lot ? $lotId->id : $lotId;

        return DB::transaction(function () use ($resolvedBuyerId, $resolvedLotId, $quantity): Order {
            /** @var Lot $lockedLot */
            $lockedLot = Lot::whereKey($resolvedLotId)->lockForUpdate()->firstOrFail();
            /** @var User $lockedBuyer */
            $lockedBuyer = User::whereKey($resolvedBuyerId)->lockForUpdate()->firstOrFail();

            if ($lockedBuyer->id === $lockedLot->seller_id) {
                throw new DomainException('Вы не можете купить собственный лот.');
            }

            if (! $lockedBuyer->status->canTrade()) {
                throw new DomainException('Ваш аккаунт не может совершать покупки.');
            }

            if ($lockedLot->status !== LotStatus::Active || $lockedLot->stock_quantity < $quantity) {
                throw new DomainException("Недостаточно товара в наличии (доступно: {$lockedLot->stock_quantity} шт.).");
            }

            $unitPrice = (float) $lockedLot->price;
            $totalPrice = round($unitPrice * $quantity, 2);

            if ((float) $lockedBuyer->wallet_balance < $totalPrice) {
                throw new DomainException('Недостаточно средств на балансе кошелька.');
            }

            $newBalance = round((float) $lockedBuyer->wallet_balance - $totalPrice, 2);
            $lockedBuyer->update(['wallet_balance' => $newBalance]);

            $newStock = $lockedLot->stock_quantity - $quantity;
            $lockedLot->update([
                'stock_quantity' => $newStock,
                'status' => $newStock === 0 ? LotStatus::Sold : LotStatus::Active,
            ]);

            $hasAutoDelivery = $lockedLot->hasAutoDelivery();

            /** @var Order $order */
            $order = Order::create([
                'lot_id' => $lockedLot->id,
                'buyer_id' => $lockedBuyer->id,
                'seller_id' => $lockedLot->seller_id,
                'quantity' => $quantity,
                'unit_price' => $unitPrice,
                'total_price' => $totalPrice,
                'status' => $hasAutoDelivery ? OrderStatus::Shipped : OrderStatus::Paid,
            ]);

            WalletTransaction::create([
                'user_id' => $lockedBuyer->id,
                'order_id' => $order->id,
                'type' => TransactionType::Purchase,
                'amount' => -$totalPrice,
                'balance_after' => $newBalance,
            ]);

            Message::create([
                'order_id' => $order->id,
                'user_id' => $lockedBuyer->id,
                'body' => "Сделка #{$order->id} оплачена ({$quantity} шт., {$totalPrice} ₽ заморожено в эскроу SportPay).",
                'is_system' => true,
            ]);

            if ($hasAutoDelivery) {
                Message::create([
                    'order_id' => $order->id,
                    'user_id' => $lockedLot->seller_id,
                    'body' => "⚡ АВТОВЫДАЧА SportPay: {$lockedLot->auto_delivery_text}",
                    'is_system' => true,
                ]);
            }

            return $order;
        });
    }

    public function confirmOrder(
        Order|int $orderId,
        User|int $actorId,
        int $rating = 5,
        ?string $comment = null
    ): Order {
        $resolvedOrderId = $orderId instanceof Order ? $orderId->id : $orderId;
        $resolvedActorId = $actorId instanceof User ? $actorId->id : $actorId;

        return DB::transaction(function () use ($resolvedOrderId, $resolvedActorId, $rating, $comment): Order {
            /** @var Order $lockedOrder */
            $lockedOrder = Order::whereKey($resolvedOrderId)->lockForUpdate()->firstOrFail();
            /** @var User $actor */
            $actor = User::whereKey($resolvedActorId)->firstOrFail();

            if ($lockedOrder->buyer_id !== $actor->id && ! $actor->isArbiter()) {
                throw new DomainException('Подтвердить выполнение заказа может только покупатель или арбитр.');
            }

            if (! $lockedOrder->status->isConfirmable()) {
                throw new DomainException('Заказ уже завершен или отменен.');
            }

            /** @var User $seller */
            $seller = User::whereKey($lockedOrder->seller_id)->lockForUpdate()->firstOrFail();

            $totalPrice = (float) $lockedOrder->total_price;
            $fee = round($totalPrice * self::PLATFORM_FEE_RATE, 2);
            $payout = round($totalPrice - $fee, 2);

            $newSellerBalance = round((float) $seller->wallet_balance + $payout, 2);
            $seller->update(['wallet_balance' => $newSellerBalance]);

            $lockedOrder->update(['status' => OrderStatus::Completed]);

            WalletTransaction::create([
                'user_id' => $seller->id,
                'order_id' => $lockedOrder->id,
                'type' => TransactionType::Payout,
                'amount' => $payout,
                'balance_after' => $newSellerBalance,
            ]);

            if ($lockedOrder->buyer_id === $actor->id) {
                $clampedRating = max(1, min(5, $rating));

                Review::updateOrCreate(
                    ['order_id' => $lockedOrder->id],
                    [
                        'lot_id' => $lockedOrder->lot_id,
                        'author_id' => $actor->id,
                        'seller_id' => $seller->id,
                        'rating' => $clampedRating,
                        'comment' => filled($comment) ? trim((string) $comment) : 'Сделка прошла отлично!',
                    ]
                );

                $avgRating = Review::where('seller_id', $seller->id)->avg('rating');
                if ($avgRating !== null) {
                    $seller->update(['rating' => round((float) $avgRating, 2)]);
                }
            }

            Message::create([
                'order_id' => $lockedOrder->id,
                'user_id' => $actor->id,
                'body' => "Заказ #{$lockedOrder->id} подтвержден! Продавцу перечислено {$payout} ₽ (комиссия биржи 5%: {$fee} ₽).",
                'is_system' => true,
            ]);

            return $lockedOrder;
        });
    }

    public function openDispute(Order|int $orderId, User|int $actorId, string $reason = 'Открыт спор'): Order
    {
        $resolvedOrderId = $orderId instanceof Order ? $orderId->id : $orderId;
        $resolvedActorId = $actorId instanceof User ? $actorId->id : $actorId;

        return DB::transaction(function () use ($resolvedOrderId, $resolvedActorId, $reason): Order {
            /** @var Order $lockedOrder */
            $lockedOrder = Order::whereKey($resolvedOrderId)->lockForUpdate()->firstOrFail();
            /** @var User $actor */
            $actor = User::whereKey($resolvedActorId)->firstOrFail();

            if ($lockedOrder->buyer_id !== $actor->id && $lockedOrder->seller_id !== $actor->id) {
                throw new DomainException('Только участники сделки могут открыть спор.');
            }

            if (! $lockedOrder->status->isDisputable()) {
                throw new DomainException('По данному заказу нельзя открыть спор.');
            }

            $lockedOrder->update(['status' => OrderStatus::Disputed]);

            Message::create([
                'order_id' => $lockedOrder->id,
                'user_id' => $actor->id,
                'body' => "⚠️ Открыт арбитражный спор пользователем {$actor->username}. Причина: {$reason}",
                'is_system' => true,
            ]);

            return $lockedOrder;
        });
    }

    public function resolveDispute(
        Order|int $orderId,
        User|int $arbiterId,
        string $decision,
        ?string $comment = null
    ): Order {
        $resolvedOrderId = $orderId instanceof Order ? $orderId->id : $orderId;
        $resolvedArbiterId = $arbiterId instanceof User ? $arbiterId->id : $arbiterId;

        /** @var Order $order */
        $order = Order::findOrFail($resolvedOrderId);
        /** @var User $arbiter */
        $arbiter = User::findOrFail($resolvedArbiterId);

        if (! $arbiter->isArbiter()) {
            throw new DomainException('Только официальный арбитр может выносить решение по спору.');
        }

        if ($order->status !== OrderStatus::Disputed) {
            throw new DomainException('Сделка не находится в статусе спора.');
        }

        $note = filled($comment) ? " Комментарий арбитра: {$comment}" : '';

        if ($decision === 'refund_buyer') {
            $resolved = $this->cancelOrder($order, $arbiter);
            Message::create([
                'order_id' => $resolved->id,
                'user_id' => $arbiter->id,
                'body' => "⚖️ Решение Арбитра: возврат 100% средств покупателю.{$note}",
                'is_system' => true,
            ]);

            return $resolved;
        }

        if ($decision === 'release_seller') {
            $resolved = $this->confirmOrder($order, $arbiter);
            Message::create([
                'order_id' => $resolved->id,
                'user_id' => $arbiter->id,
                'body' => "⚖️ Решение Арбитра: выплата средств продавцу.{$note}",
                'is_system' => true,
            ]);

            return $resolved;
        }

        throw new DomainException('Неизвестное решение арбитра.');
    }

    public function cancelOrder(Order|int $orderId, User|int $actorId): Order
    {
        $resolvedOrderId = $orderId instanceof Order ? $orderId->id : $orderId;
        $resolvedActorId = $actorId instanceof User ? $actorId->id : $actorId;

        return DB::transaction(function () use ($resolvedOrderId, $resolvedActorId): Order {
            /** @var Order $lockedOrder */
            $lockedOrder = Order::whereKey($resolvedOrderId)->lockForUpdate()->firstOrFail();
            /** @var User $actor */
            $actor = User::whereKey($resolvedActorId)->firstOrFail();

            if ($lockedOrder->seller_id !== $actor->id && ! $actor->isArbiter()) {
                throw new DomainException('Отменить оплаченный заказ и вернуть средства может только продавец или арбитр.');
            }

            if (! $lockedOrder->status->isConfirmable()) {
                throw new DomainException('Заказ уже завершен или отменен.');
            }

            /** @var User $buyer */
            $buyer = User::whereKey($lockedOrder->buyer_id)->lockForUpdate()->firstOrFail();
            /** @var Lot|null $lot */
            $lot = Lot::whereKey($lockedOrder->lot_id)->lockForUpdate()->first();

            $refundAmount = (float) $lockedOrder->total_price;
            $newBuyerBalance = round((float) $buyer->wallet_balance + $refundAmount, 2);
            $buyer->update(['wallet_balance' => $newBuyerBalance]);

            if ($lot !== null) {
                $restoredStock = $lot->stock_quantity + $lockedOrder->quantity;
                $lot->update([
                    'stock_quantity' => $restoredStock,
                    'status' => LotStatus::Active,
                ]);
            }

            $lockedOrder->update(['status' => OrderStatus::Cancelled]);

            WalletTransaction::create([
                'user_id' => $buyer->id,
                'order_id' => $lockedOrder->id,
                'type' => TransactionType::Deposit,
                'amount' => $refundAmount,
                'balance_after' => $newBuyerBalance,
            ]);

            Message::create([
                'order_id' => $lockedOrder->id,
                'user_id' => $actor->id,
                'body' => "Заказ #{$lockedOrder->id} отменен. Покупателю возвращено {$refundAmount} ₽, товар ({$lockedOrder->quantity} шт.) возвращен на склад.",
                'is_system' => true,
            ]);

            return $lockedOrder;
        });
    }
}

namespace App\Http\Controllers;

use App\Events\OrderMessageSent;

use Illuminate\Contracts\View\View;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\RedirectResponse;

class ChatController extends Controller
{
    public function index(Request $request): View
    {
        $user = $this->currentUser();

        $orders = Order::with(['lot', 'buyer', 'seller', 'latestMessage'])
            ->where(function ($query) use ($user): void {
                $query->where('buyer_id', $user->id)
                    ->orWhere('seller_id', $user->id);
            })
            ->withCount([
                'messages as unread_count' => function ($query) use ($user): void {
                    $query->where('user_id', '!=', $user->id)->whereNull('read_at');
                },
            ])
            ->latest('updated_at')
            ->get();

        $activeOrder = null;

        if ($request->filled('order_id')) {
            $activeOrder = Order::with(['lot', 'buyer', 'seller', 'messages.user'])
                    ->findOrFail((int) $request->integer('order_id'));

            $this->authorizeAction('view', $activeOrder);
        } elseif ($orders->isNotEmpty()) {
            $activeOrder = Order::with(['lot', 'buyer', 'seller', 'messages.user'])
                    ->find($orders->first()->id);
        }

        if ($activeOrder !== null) {
            Message::where('order_id', $activeOrder->id)
                ->where('user_id', '!=', $user->id)
                ->whereNull('read_at')
                ->update(['read_at' => now()]);
        }

        return view('chat.index', [
            'user' => $user,
            'allUsers' => User::orderBy('id')->get(),
            'orders' => $orders,
            'activeOrder' => $activeOrder,
        ]);
    }

    public function messages(Request $request, Order$order): JsonResponse
    {
        $user = $this->currentUser();
        $this->authorizeAction('view', $order);

        $afterId = (int)$request->query('after_id', 0);

        $messages = Message::with('user')
            ->where('order_id', $order->id)
            ->where('id', '>', $afterId)
            ->oldest('id')
            ->get();

        if ($messages->isNotEmpty()) {
            Message::where('order_id', $order->id)
                ->where('user_id', '!=', $user->id)
                ->whereNull('read_at')
                ->update(['read_at' => now()]);
        }

        return response()->json([
            'messages' => $messages->map(fn (Message$msg): array => [
                'id' => $msg->id,
                'order_id' => $msg->order_id,
                'user_id' => $msg->user_id,
                'username' => $msg->user->username ?? 'Система',
                'body' => $msg->body,
                'is_system' => (bool) $msg->is_system,
                'time' => $msg->created_at?->format('H:i') ?? now()->format('H:i'),
            ]),
        ]);
    }

    public function store(Request $request, Order$order): JsonResponse|RedirectResponse
    {
        $user = $this->currentUser();
        $this->authorizeAction('sendMessage', $order);

        $data = $request->validate([
            'body' => ['required', 'string', 'max:2000'],
        ]);

        $message = Message::create([
            'order_id' => $order->id,
            'user_id' => $user->id,
            'body' => $data['body'],
            'is_system' => false,
        ]);

        $message->load('user');
        $order->touch();

        try {
            broadcast(new OrderMessageSent($message))->toOthers();
        } catch (\Throwable) {
            // Игнорируем ошибку сокета, если Reverb не запущен — отработает фоновый пуллинг
        }

        $payload = [
            'id' => $message->id,
            'order_id' => $order->id,
            'user_id' => $user->id,
            'username' => $user->username,
            'body' => $message->body,
            'is_system' => false,
            'time' => $message->created_at?->format('H:i') ?? now()->format('H:i'),
        ];

        if ($request->wantsJson()) {
            return response()->json([
                'status' => 'success',
                'message' => $payload,
            ]);
        }

        return back();
    }
}

     */

    {
        $request->validate([
            'token' => 'required',
            'email' => 'required|email',
            'password' => ['required', 'confirmed', Rules\Password::defaults()],
        ]);

        // Here we will attempt to reset the user's password. If it is successful we

        // database. Otherwise we will parse the error and return the response.
        $status = Password::reset(
            $request->only('email', 'password', 'password_confirmation', 'token'),
            function ($user) use ($request) {

                    'remember_token' => Str::random(60),

                event(new PasswordReset($user));
            }
        );

        // If the password was successfully reset, we will redirect the user back to
        // the application's home authenticated view. If there is an error we can
        // redirect them back to where they came from with their error message.
        if ($status == Password::PASSWORD_RESET) {
            return redirect()->route('login')->with('status', __($status));
        }

        throw ValidationException::withMessages([
            'email' => [trans($status)],
        ]);
    }

==> app/Http/Requests/Auth/LoginRequest.php <==
<?php

declare(strict_types=1);

namespace App\Http\Requests\Auth;

use Illuminate\Auth\Events\Lockout;
use Illuminate\Contracts\Validation\ValidationRule;
use Illuminate\Foundation\Http\FormRequest;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;

class LoginRequest extends FormRequest
{
    /**
     * Determine if the user is authorized to make this request.
     */
    public function authorize(): bool
    {
        return true;
    }

    /**
     * Get the validation rules that apply to the request.
     *
     * @return array<string, ValidationRule|array<mixed>|string>
     */
    public function rules(): array
    {
        return [
            'email' => ['required', 'string', 'email'],
            'password' => ['required', 'string'],
        ];
    }

    /**
     * Attempt to authenticate the request's credentials.
     *
     * @throws ValidationException
     */
    public function authenticate(): void
    {
        $this->ensureIsNotRateLimited();

        if (! Auth::attempt($this->only('email', 'password'), $this->boolean('remember'))) {
            RateLimiter::hit($this->throttleKey());

            throw ValidationException::withMessages([
                'email' => trans('auth.failed'),
            ]);
        }

        RateLimiter::clear($this->throttleKey());
    }

    /**
     * Ensure the login request is not rate limited.
     *
     * @throws ValidationException
     */
    public function ensureIsNotRateLimited(): void
    {
        if (! RateLimiter::tooManyAttempts($this->throttleKey(), 5)) {
            return;
        }

        event(new Lockout($this));

        $seconds = RateLimiter::availableIn($this->throttleKey());

        throw ValidationException::withMessages([
            'email' => trans('auth.throttle', [
                'seconds' => $seconds,
                'minutes' => ceil($seconds / 60),
            ]),
        ]);
    }

    /**
     * Get the rate limiting throttle key for the request.
     */
    public function throttleKey(): string
    {
        return Str::transliterate(Str::lower($this->string('email')).'|'.$this->ip());
    }
}

<?php

declare(strict_types=1);

namespace App\Models;

use App\Enums\LotStatus;
use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Support\Carbon;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;

class Lot extends Model
{
    use HasFactory;

    protected $fillable = [
        'seller_id',
        'title',
        'product_type',
        'description',
        'location_name',
        'latitude',
        'longitude',
        'image_path',
        'auto_delivery_text',
        'price',
        'stock_quantity',
        'status',
    ];

    protected $hidden = [
        'auto_delivery_text',
        'coordinates',
    ];

    protected static function booted(): void
    {
        static::saved(function (Lot $lot): void {
            if (DB::getDriverName() !== 'pgsql') {
                return;
            }

            if ($lot->latitude !== null && $lot->longitude !== null) {
                DB::statement(
                    'UPDATE lots SET coordinates = ST_SetSRID(ST_MakePoint(?, ?), 4326)::geography WHERE id = ?',
                    [(float) $lot->longitude, (float) $lot->latitude, $lot->id]
                );
            } else {
                DB::statement('UPDATE lots SET coordinates = NULL WHERE id = ?', [$lot->id]);
            }
        });
    }

    public function hasAutoDelivery(): bool
    {
        return filled($this->auto_delivery_text);
    }

    public function hasLocation(): bool
    {
        return $this->latitude !== null && $this->longitude !== null;
    }

    public function imageUrl(): ?string
    {
        if (! filled($this->image_path)) {
            return null;
        }

        return Storage::disk('public')->url($this->image_path);
    }

    protected function casts(): array
    {
        return [
            'price' => 'decimal:2',
            'latitude' => 'decimal:7',
            'longitude' => 'decimal:7',
            'stock_quantity' => 'integer',
            'status' => LotStatus::class,
        ];
    }

    public function seller(): BelongsTo
    {
        return $this->belongsTo(User::class, 'seller_id');
    }

    public function orders(): HasMany
    {
        return $this->hasMany(Order::class, 'lot_id');
    }

    public function reviews(): HasMany
    {
        return $this->hasMany(Review::class, 'lot_id');
    }

    public function scopeFilter(
        Builder $query,
        ?string $type = null,
        ?string $search = null,
        ?float $minPrice = null,
        ?float $maxPrice = null,
        ?string $seller = null,
        bool $onlineOnly = false,
        ?float $userLat = null,
        ?float $userLng = null,
        ?float $radiusKm = null
    ): Builder {
        $likeOperator = DB::getDriverName() === 'pgsql' ? 'ilike' : 'like';

        $query->select('lots.*')
            ->where('lots.stock_quantity', '>', 0)
            ->where('lots.status', LotStatus::Active)
            ->when($type, fn (Builder $q) => $q->where('lots.product_type', $type))
            ->when($search, function (Builder $q) use ($search, $likeOperator): void {
                $term = '%'.trim($search).'%';
                $q->where(function (Builder $sub) use ($term, $likeOperator): void {
                    $sub->where('lots.title', $likeOperator, $term)
                        ->orWhere('lots.description', $likeOperator, $term)
                        ->orWhere('lots.location_name', $likeOperator, $term);
                });
            })
            ->when($minPrice !== null && $minPrice > 0, fn (Builder $q) => $q->where('lots.price', '>=', $minPrice))
            ->when($maxPrice !== null && $maxPrice > 0, fn (Builder $q) => $q->where('lots.price', '<=', $maxPrice))
            ->when($seller, function (Builder $q) use ($seller, $likeOperator): void {
                $cleanSeller = trim($seller);
                $q->whereHas('seller', function (Builder $sq) use ($cleanSeller, $likeOperator): void {
                    if (is_numeric($cleanSeller)) {
                        $sq->where('users.id', (int) $cleanSeller);
                    } else {
                        $sq->where('users.username', $likeOperator, '%'.$cleanSeller.'%');
                    }
                });
            })
            ->when($onlineOnly, function (Builder $q): void {
                $q->whereHas('seller', fn (Builder $sq) => $sq->where('users.last_seen_at', '>=', now()->subMinutes(5)));
            });

        if (DB::getDriverName() === 'pgsql' && $userLat !== null && $userLng !== null) {
            $query->selectRaw(
                'ROUND((ST_Distance(lots.coordinates, ST_SetSRID(ST_MakePoint(?, ?), 4326)::geography) / 1000.0)::numeric, 1) as distance_km',
                [$userLng, $userLat]
            );

            if ($radiusKm !== null && $radiusKm > 0) {
                $query->whereNotNull('lots.coordinates')
                    ->whereRaw(
                        'ST_DWithin(lots.coordinates, ST_SetSRID(ST_MakePoint(?, ?), 4326)::geography, ?)',
                        [$userLng, $userLat, $radiusKm * 1000.0]
                    );
            }
        }

        return $query;
    }

    public function scopeSortBy(Builder $query, ?string $sort, ?float $userLat = null, ?float $userLng = null): Builder
    {
        $threshold = now()->subMinutes(5)->toDateTimeString();

        if ($sort === 'nearest' && DB::getDriverName() === 'pgsql' && $userLat !== null && $userLng !== null) {
            return $query
                ->orderByRaw('lots.coordinates IS NULL ASC')
                ->orderByRaw(
                    'ST_Distance(lots.coordinates, ST_SetSRID(ST_MakePoint(?, ?), 4326)::geography) ASC',
                    [$userLng, $userLat]
                )
                ->orderByDesc('lots.id');
        }

        return match ($sort) {
            'price_asc' => $query->orderBy('lots.price', 'asc'),
            'price_desc' => $query->orderBy('lots.price', 'desc'),
            'rating_desc' => $query
                ->join('users', 'lots.seller_id', '=', 'users.id')
                ->orderByDesc('users.rating')
                ->orderByDesc('lots.id'),
            'online_first' => $query
                ->join('users', 'lots.seller_id', '=', 'users.id')
                ->orderByRaw('CASE WHEN users.last_seen_at >= ? THEN 1 ELSE 0 END DESC', [$threshold])
                ->orderByDesc('users.last_seen_at')
                ->orderByDesc('lots.id'),
            default => $query->latest('lots.id'),
        };
    }
}

async function openSellerModal(sellerId) {
    const backdrop = document.getElementById('seller-modal-backdrop');
    const reviewsBox = document.getElementById('modal-seller-reviews');
    const avatarImg = document.getElementById('modal-seller-avatar');
    const initialBox = document.getElementById('modal-seller-initial');
    const onlineDot = document.getElementById('modal-seller-online-dot');
    const onlineBadge = document.getElementById('modal-seller-online-badge');

    if (!backdrop || !reviewsBox) return;

    backdrop.classList.remove('hidden');
    document.getElementById('modal-seller-username').textContent = '...';
    reviewsBox.innerHTML = '<div class="text-slate-500 py-4 text-center">...</div>';

    try {
        const res = await fetch(`/users/${sellerId}/modal`, {
            headers: { 'Accept': 'application/json' }
        });
        if (!res.ok) throw new Error('Error');
        const data = await res.json();

        if (data.avatar_url) {
            avatarImg.src = data.avatar_url;
            avatarImg.classList.remove('hidden');
            initialBox.classList.add('hidden');
        } else {
            avatarImg.classList.add('hidden');
            initialBox.classList.remove('hidden');
            initialBox.textContent = data.initial;
        }

        onlineDot.className = `absolute -bottom-1 -right-1 w-3.5 h-3.5 rounded-full border-2 border-slate-900 ${data.is_online ? 'bg-emerald-400' : 'bg-slate-600'}`;
        onlineBadge.textContent = (data.is_online ? '🟢 ' : '⚪ ') + data.online_label;
        onlineBadge.className = data.is_online
            ? 'text-[11px] font-bold px-2 py-0.5 rounded-full bg-emerald-500/20 text-emerald-300 border border-emerald-500/30'
            : 'text-[11px] font-bold px-2 py-0.5 rounded-full bg-slate-800 text-slate-400 border border-slate-700';

        document.getElementById('modal-seller-username').textContent = data.username;
        document.getElementById('modal-seller-status').textContent = data.status_label;
        document.getElementById('modal-seller-registered').textContent = `SportPay • ${data.registered_at}`;
        document.getElementById('modal-seller-rating').textContent = `★ ${data.rating}`;
        document.getElementById('modal-seller-orders').textContent = data.completed_orders;
        document.getElementById('modal-seller-lots').textContent = data.active_lots_count;
        document.getElementById('modal-seller-filter-link').href = data.filter_seller_url;
        document.getElementById('modal-seller-profile-link').href = data.profile_url;

        if (!data.reviews || data.reviews.length === 0) {
            reviewsBox.innerHTML = '<div class="text-slate-500 py-4 text-center">—</div>';
        } else {
            reviewsBox.innerHTML = '';
            data.reviews.forEach(rev => {
                const item = document.createElement('div');
                item.className = 'p-3 rounded-xl bg-slate-950 border border-slate-800 space-y-1';
                const stars = '★'.repeat(Math.max(1, Math.min(5, Number(rev.rating || 5))));
                item.innerHTML = `
                    <div class="flex items-center justify-between">
                        <span class="font-bold text-emerald-400"></span>
                        <span class="text-amber-400 font-bold">${stars}</span>
                    </div>
                    <div class="text-[11px] text-slate-400"></div>
                    <div class="text-slate-200"></div>
                `;
                item.querySelector('.text-emerald-400').textContent = rev.author;
                item.querySelector('.text-slate-400').textContent = `${rev.lot_title} • ${rev.date}`;
                item.querySelector('.text-slate-200').textContent = rev.comment;
                reviewsBox.appendChild(item);
            });
        }
    } catch (e) {
        reviewsBox.innerHTML = '<div class="text-rose-400 py-4 text-center">Error</div>';
    }
}

(function () {
    const box = document.getElementById('order-messages-box');
    const form = document.getElementById('order-chat-form');
    const input = document.getElementById('order-chat-input');
    const currentUserId = {{ (int) $user->id }};
    const orderId = {{ (int) $order->id }};
    const pollUrl = "{{ route('orders.messages.index', $order) }}";

    let lastMessageId = {{ (int) ($order->messages->last()?->id ?? 0) }};

    function forceScrollBottom() {
        if (!box) return;
        box.scrollTop = box.scrollHeight + 9999;
    }

    forceScrollBottom();
    window.addEventListener('load', () => {
        forceScrollBottom();
        setTimeout(forceScrollBottom, 120);
    });

    function renderOrderMessage(msg) {
        const msgId = Number(msg.id || 0);
        if (msgId > 0 && box.querySelector(`[data-msg-id="${msgId}"]`)) {
            return;
        }
        if (msgId > lastMessageId) {
            lastMessageId = msgId;
        }

        const emptyPlaceholder = document.getElementById('empty-order-chat');
        if (emptyPlaceholder) emptyPlaceholder.remove();

        const wrapper = document.createElement('div');
        if (msgId > 0) wrapper.setAttribute('data-msg-id', String(msgId));

        if (msg.is_system) {
            wrapper.className = 'p-3 rounded-xl bg-amber-500/10 border border-amber-500/20 text-center text-xs text-amber-300';
            wrapper.textContent = '🛡️ ' + msg.body;
        } else {
            const isMe = Number(msg.user_id) === currentUserId;
            wrapper.className = `flex flex-col ${isMe ? 'items-end' : 'items-start'}`;

            const meta = document.createElement('div');
            meta.className = 'text-[11px] text-slate-400 mb-1';
            meta.textContent = `${msg.username} • ${msg.time}`;

            const bubble = document.createElement('div');
            bubble.className = `max-w-[85%] sm:max-w-md px-3.5 sm:px-4 py-2.5 rounded-2xl text-sm ${isMe ? 'bg-emerald-500 text-slate-950 font-medium' : 'bg-slate-800 text-slate-100 border border-slate-700'}`;
            bubble.textContent = msg.body;

            wrapper.appendChild(meta);
            wrapper.appendChild(bubble);
        }

        box.appendChild(wrapper);
        forceScrollBottom();
    }

    form.addEventListener('submit', async function (e) {
        e.preventDefault();
        const body = input.value.trim();
        if (!body) return;

        input.value = '';
        input.focus();

        try {
            const response = await fetch(form.action, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'X-CSRF-TOKEN': '{{ csrf_token() }}',
                    'Accept': 'application/json'
                },
                body: JSON.stringify({ body })
            });

            if (response.ok) {
                const data = await response.json();
                if (data.message) renderOrderMessage(data.message);
            }
        } catch (err) {
            console.error('Error:', err);
        }
    });

    if (window.Echo) {
        window.Echo.channel(`orders.${orderId}`)
            .listen('.order.message.sent', (data) => renderOrderMessage(data));
    }

    setInterval(async () => {
        try {
            const res = await fetch(`${pollUrl}?after_id=${lastMessageId}`, {
                headers: { 'Accept': 'application/json' }
            });
            if (res.ok) {
                const data = await res.json();
                if (Array.isArray(data.messages)) {
                    data.messages.forEach(renderOrderMessage);
                }
            }
        } catch (e) {}
    }, 2000);
})();

<script>
(function () {
    const chatMessages = document.getElementById('chat-messages');
    const chatForm = document.getElementById('chat-form');
    const chatInput = document.getElementById('chat-input');
    const currentUserId = {{ (int) $user->id }};
    const pollUrl = "{{ route('chat.global.messages') }}";

    let lastMessageId = {{ (int) ($messages->last()?->id ?? 0) }};

    function forceScrollBottom() {
        if (!chatMessages) return;
        chatMessages.scrollTop = chatMessages.scrollHeight + 9999;
    }

    forceScrollBottom();
    window.addEventListener('load', () => {
        forceScrollBottom();
        setTimeout(forceScrollBottom, 120);
    });

    function appendGlobalMessage(msg) {
        const msgId = Number(msg.id || 0);
        if (msgId > 0 && chatMessages.querySelector(`[data-msg-id="${msgId}"]`)) {
            return;
        }
        if (msgId > lastMessageId) {
            lastMessageId = msgId;
        }

        const noMsg = document.getElementById('no-messages');
        if (noMsg) noMsg.remove();

        const isMe = Number(msg.user_id) === currentUserId;
        const wrapper = document.createElement('div');
        wrapper.className = `flex flex-col ${isMe ? 'items-end' : 'items-start'}`;
        if (msgId > 0) wrapper.setAttribute('data-msg-id', String(msgId));

        const meta = document.createElement('div');
        meta.className = 'text-[11px] text-slate-400 mb-1';
        meta.innerHTML = `<span class="font-bold text-slate-300"></span> • ${msg.time}`;
        meta.querySelector('span').textContent = msg.username;

        const bubble = document.createElement('div');
        bubble.className = `max-w-[85%] sm:max-w-lg px-3.5 sm:px-4 py-2.5 rounded-2xl text-sm ${isMe ? 'bg-emerald-500 text-slate-950 font-medium' : 'bg-slate-800 text-slate-100 border border-slate-700'}`;
        bubble.textContent = msg.body;

        wrapper.appendChild(meta);
        wrapper.appendChild(bubble);

        chatMessages.appendChild(wrapper);
        forceScrollBottom();
    }

    chatForm.addEventListener('submit', async function (e) {
        e.preventDefault();
        const body = chatInput.value.trim();
        if (!body) return;

        chatInput.value = '';
        chatInput.focus();

        try {
            const response = await fetch("{{ route('chat.global.store') }}", {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'X-CSRF-TOKEN': '{{ csrf_token() }}',
                    'Accept': 'application/json'
                },
                body: JSON.stringify({ body })
            });

            if (response.ok) {
                const data = await response.json();
                if (data.message) appendGlobalMessage(data.message);
            }
        } catch (err) {
            console.error('Error:', err);
        }
    });

    if (window.Echo) {
        window.Echo.channel('public-chat')
            .listen('.message.sent', (data) => appendGlobalMessage(data));
    }

    setInterval(async () => {
        try {
            const res = await fetch(`${pollUrl}?after_id=${lastMessageId}`, {
                headers: { 'Accept': 'application/json' }
            });
            if (res.ok) {
                const data = await res.json();
                if (Array.isArray(data.messages)) {
                    data.messages.forEach(appendGlobalMessage);
                }
            }
        } catch (e) {}
    }, 2000);
})();
